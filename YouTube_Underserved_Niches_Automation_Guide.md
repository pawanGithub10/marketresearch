**Underserved Opportunities (100-500 channels):**

- Often HIGHER CPM than saturated niches

    def score_niche(self, keyword_data: Dict) -> int:
        """
        Score niche from 0-100 based on opportunity metrics
        """
        metrics = keyword_data['metrics']
        score = 0
        reasoning = []

        # Search Volume Score (25 points max)
        search_vol = metrics.get('estimated_search_volume', 0)
        if search_vol > 100000:
            score += 25
            reasoning.append("✓ Excellent search volume (100k+)")
        elif search_vol > 50000:
            score += 20
            reasoning.append("✓ Good search volume (50k-100k)")
        elif search_vol > 20000:
            score += 15
            reasoning.append("✓ Moderate search volume (20k-50k)")
        elif search_vol > 10000:
            score += 10
            reasoning.append("⚠ Lower search volume (10k-20k)")
        else:
            score += 5
            reasoning.append("✗ Low search volume (<10k)")

        # Difficulty/Competition Score (20 points max)
        difficulty = metrics.get('estimated_difficulty', 100)
        if difficulty < 30:
            score += 20
            reasoning.append("✓ Low competition (KD < 30)")
        elif difficulty < 50:
            score += 15
            reasoning.append("✓ Moderate competition (KD 30-50)")
        elif difficulty < 75:
            score += 10
            reasoning.append("⚠ High competition (KD 50-75)")
        else:
            score += 5
            reasoning.append("✗ Very high competition (KD > 75)")

        # CPM Score (20 points max)
        cpm = metrics.get('estimated_cpm', 0)
        if cpm >= 15:
            score += 20
            reasoning.append("✓ Excellent CPM ($15+)")
        elif cpm >= 8:
            score += 15
            reasoning.append("✓ Good CPM ($8-15)")
        elif cpm >= 4:
            score += 10
            reasoning.append("✓ Moderate CPM ($4-8)")
        elif cpm >= 2:
            score += 5
            reasoning.append("⚠ Low CPM ($2-4)")
        else:
            score += 2
            reasoning.append("✗ Very low CPM (<$2)")

        # Monetization Potential (15 points max)
        monetization_bonus = metrics.get('monetization_potential', 0)
        score += monetization_bonus
        if monetization_bonus > 10:
            reasoning.append("✓ Multiple monetization streams")

        # Niche Longevity (10 points max)
        trend_direction = metrics.get('trend_direction', 'stable')
        if trend_direction == 'growing':
            score += 10
            reasoning.append("✓ Growing trend")
        elif trend_direction == 'stable':
            score += 7
            reasoning.append("✓ Stable market")
        else:
            score += 3
            reasoning.append("✗ Declining trend")

        keyword_data['metrics']['opportunity_score'] = score
        keyword_data['reasoning'] = reasoning

        # Generate recommendation
        if score >= 85:
            keyword_data['recommendation'] = "🟢 EXCELLENT - Launch immediately"
        elif score >= 70:
            keyword_data['recommendation'] = "🟢 GOOD - Proceed with confidence"
        elif score >= 55:
            keyword_data['recommendation'] = "🟡 FAIR - Viable but needs differentiation"
        else:
            keyword_data['recommendation'] = "🔴 POOR - Reconsider this niche"

        return score

    def analyze_multiple_niches(self, keywords: List[str]) -> List[Dict]:
        """
        Analyze multiple niche keywords and rank them
        """
        print(f"\n🔍 Analyzing {len(keywords)} potential niches...\n")

        results = []
        for keyword in keywords:
            result = self.validate_keyword(keyword)
            self.score_niche(result)
            results.append(result)

            print(f"  {keyword}")
            print(f"  Score: {result['metrics']['opportunity_score']}/100")
            print(f"  Recommendation: {result['recommendation']}\n")

        # Sort by opportunity score
        results.sort(
            key=lambda x: x['metrics']['opportunity_score'],
            reverse=True
        )

        return results

    def save_results(self, results: List[Dict], format: str = 'json'):
        """Save analysis results to file"""
        filename = f"niche_analysis_{self.timestamp}.{format}"

                        'difficulty': result['metrics']['estimated_difficulty'],
                        'recommendation': result['recommendation']
                    })

        print(f"✅ Results saved to {filename}")
        return filename

#!/usr/bin/env python3
"""
YouTube Content Calendar & Topic Generator
Generates 30 video topics with SEO optimization
"""

import json
from datetime import datetime, timedelta
from typing import List, Dict

class ContentCalendarGenerator:
    """Generate optimized video content calendar"""

    def __init__(self, niche: str, upload_frequency: int = 3):
        """
        Initialize content calendar generator

        Args:
            niche: YouTube niche topic
            upload_frequency: Videos per week (default: 3)
        """
        self.niche = niche
        self.upload_frequency = upload_frequency
        self.start_date = datetime.now()
        self.calendar = []

    def generate_video_topics(self) -> List[Dict]:
        """
        Generate 30 video topics based on niche
        In production, use ChatGPT API or similar
        """

        # Content pillar distribution
        pillars = {
            'How-To Tutorial': 0.30,  # 30% of videos
            'Reviews & Comparisons': 0.25,  # 25%
            'Beginner Guide': 0.20,  # 20%
            'Advanced Tips': 0.15,  # 15%
            'Trending Topic': 0.10  # 10%
        }

        # Example topics for "AI Tools for Dentists" niche
        topic_templates = {
            'How-To Tutorial': [
                f"How to Use ChatGPT for {self.niche}: Step-by-Step",
                f"Automating Appointment Scheduling with AI",
                f"Using AI for Patient Consultation Notes",
                f"AI-Powered Dental Practice Management",
                f"ChatGPT Prompts Every {self.niche} Professional Needs",
            ],
            'Reviews & Comparisons': [
                f"Best AI Tools for {self.niche}: Complete Comparison",
                f"{self.niche} Software vs AI Automation: Which Wins?",
                f"Top 5 AI Tools That Save {self.niche} Time",
                f"Comparing Dental AI Tools: Full Review",
            ],
            'Beginner Guide': [
                f"What is AI and How Does It Help {self.niche}?",
                f"Complete Beginner's Guide to AI Tools",
                f"No Technical Skills? Start Using AI Today",
            ],
            'Advanced Tips': [
                f"Advanced AI Workflows for {self.niche} Automation",
                f"Combining Multiple AI Tools for Maximum Efficiency",
            ],
            'Trending Topic': [
                f"New AI Tool That {self.niche} Is Using",
                f"OpenAI's Latest Update: What It Means for {self.niche}",
            ]
        }

        topics = []
        topic_count = 0

        for pillar, percentage in pillars.items():
            count = int(30 * percentage)
            pillar_topics = topic_templates.get(pillar, [])

            for i in range(count):
                topic = {
                    'id': topic_count + 1,
                    'title': pillar_topics[i % len(pillar_topics)] if pillar_topics else f"{pillar} #{i+1}",
                    'pillar': pillar,
                    'publish_date': self.start_date + timedelta(days=topic_count*2),  # Every 2 days
                    'estimated_length': '6-8 minutes',
                    'seo_score': 0,
                    'affiliate_potential': 'High',
                    'status': 'Planning'
                }
                topics.append(topic)
                topic_count += 1

        return topics

    def calculate_seo_score(self, title: str) -> int:
        """
        Calculate basic SEO score for video title (0-100)
        """
        score = 0

        # Check for keyword relevance
        if self.niche.lower() in title.lower():
            score += 20

        # Check for power words
        power_words = ['complete', 'ultimate', 'guide', 'how to', 'best', 'top', 'easy']
        if any(word in title.lower() for word in power_words):
            score += 15

        # Check title length (optimal: 50-60 chars)
        if 50 <= len(title) <= 60:
            score += 15
        elif 40 <= len(title) <= 70:
            score += 10

        # Check for numbers (performs well)
        if any(char.isdigit() for char in title):
            score += 15

        # Check for question format
        if title.endswith('?'):
            score += 10

        # Check for bracket inclusion
        if '[' in title or '(' in title:
            score += 10

        return min(score, 100)

    def generate_calendar(self) -> List[Dict]:
        """
        Generate complete 30-day content calendar
        """
        topics = self.generate_video_topics()

        calendar = []
        current_date = self.start_date

        for i, topic in enumerate(topics):
            topic['seo_score'] = self.calculate_seo_score(topic['title'])
            topic['publish_date'] = current_date.strftime("%Y-%m-%d %A")

            # Calculate publish time (optimize for audience timezone)
            publish_time = "2:30 PM" if (current_date.weekday() < 5) else "9:30 AM"
            topic['publish_time'] = publish_time

            calendar.append(topic)

            # Increment date based on upload frequency
            days_between = int(7 / self.upload_frequency)
            current_date += timedelta(days=days_between)

        return calendar

    def export_calendar(self, format: str = 'json') -> str:
        """
        Export calendar to file
        """
        calendar = self.generate_calendar()
        timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
        filename = f"content_calendar_{self.niche.replace(' ', '_')}_{timestamp}.{format}"

        if format == 'json':
            with open(filename, 'w') as f:
                json.dump(calendar, f, indent=2)
        elif format == 'csv':
            import csv
            with open(filename, 'w', newline='') as f:
                writer = csv.DictWriter(
                    f,
                    fieldnames=['id', 'title', 'pillar', 'publish_date', 'publish_time', 'seo_score', 'status']
                )
                writer.writeheader()
                writer.writerows(calendar)

        print(f"✅ Content calendar exported to {filename}")
        return filename

    def print_calendar(self):

#!/usr/bin/env python3
"""
YouTube Affiliate Link Manager
Organize, track, and manage affiliate links with metrics
"""

import json
import csv
from datetime import datetime
from typing import Dict, List, Optional

class AffiliateManager:
    """Manage affiliate links, programs, and earnings tracking"""

    def __init__(self):
        self.programs = []
        self.links = []
        self.earnings = []
        self.timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")

    def add_program(self, name: str, url: str, commission_rate: float,
                   payment_model: str = 'cps') -> Dict:
        """
        Add affiliate program

        Args:
            name: Program name
            url: Program URL
            commission_rate: Commission percentage (e.g., 30 for 30%)
            payment_model: 'cps' (cost per sale), 'cpc' (cost per click), 'cpl' (cost per lead)
        """

        program = {
            'id': len(self.programs) + 1,
            'name': name,
            'url': url,
            'commission_rate': commission_rate,
            'payment_model': payment_model,
            'date_joined': datetime.now().strftime("%Y-%m-%d"),
            'status': 'Active',
            'notes': ''
        }

        self.programs.append(program)
        print(f"✅ Added program: {name} ({commission_rate}% commission)")
        return program

    def create_affiliate_link(self, program_id: int, product_name: str,
                            product_url: str, video_id: str = '') -> Dict:
        """
        Create and track affiliate link
        """
        program = next((p for p in self.programs if p['id'] == program_id), None)

        if not program:
            print(f"❌ Program ID {program_id} not found")
            return {}

        link = {
            'id': len(self.links) + 1,
            'program_id': program_id,
            'program_name': program['name'],
            'product_name': product_name,
            'product_url': product_url,
            'video_id': video_id,
            'affiliate_link': self._generate_tracking_link(product_url),
            'created_date': datetime.now().strftime("%Y-%m-%d"),
            'clicks': 0,
            'conversions': 0,
            'revenue': 0.0,
            'notes': ''
        }

        self.links.append(link)
        print(f"✅ Created affiliate link for {product_name}")
        return link

    def _generate_tracking_link(self, base_url: str) -> str:
        """
        Generate trackable affiliate link with unique ID
        (In production, integrate with actual affiliate API)
        """
        link_id = f"yt_{len(self.links)+1}_{self.timestamp.replace(' ', '_')}"

        separator = '&' if '?' in base_url else '?'
        tracking_url = f"{base_url}{separator}utm_source=youtube&utm_medium=affiliate&utm_campaign={link_id}"

        return tracking_url

    def log_conversion(self, link_id: int, clicks: int = 1,
                      conversions: int = 1, revenue: float = 0.0):
        """
        Log conversion for affiliate link
        """
        link = next((l for l in self.links if l['id'] == link_id), None)

        if not link:
            print(f"❌ Link ID {link_id} not found")
            return

        link['clicks'] += clicks
        link['conversions'] += conversions
        link['revenue'] += revenue

        # Log to earnings history
        earning_record = {
            'link_id': link_id,
            'program_name': link['program_name'],
            'product_name': link['product_name'],
            'clicks': clicks,
            'conversions': conversions,
            'revenue': revenue,
            'date': datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        }
        self.earnings.append(earning_record)

        print(f"✅ Logged conversion for {link['product_name']}: ${revenue} revenue")

    def get_link_performance(self, link_id: int) -> Dict:
        """
        Get performance metrics for specific link
        """
        link = next((l for l in self.links if l['id'] == link_id), None)

        if not link:
            return {}

        conversion_rate = (link['conversions'] / link['clicks'] * 100) if link['clicks'] > 0 else 0
        revenue_per_click = link['revenue'] / link['clicks'] if link['clicks'] > 0 else 0

        return {
            'product': link['product_name'],
            'clicks': link['clicks'],
            'conversions': link['conversions'],
            'conversion_rate': f"{conversion_rate:.2f}%",
            'revenue': f"${link['revenue']:.2f}",
            'revenue_per_click': f"${revenue_per_click:.2f}",
            'ctr': f"{(link['clicks']/1000):.2f}%" if link['clicks'] > 0 else "0%"
        }

    def get_program_summary(self, program_id: int) -> Dict:
        """
        Get summary for affiliate program
        """
        program = next((p for p in self.programs if p['id'] == program_id), None)

        if not program:
            return {}

        program_links = [l for l in self.links if l['program_id'] == program_id]
        total_clicks = sum(l['clicks'] for l in program_links)
        total_conversions = sum(l['conversions'] for l in program_links)
        total_revenue = sum(l['revenue'] for l in program_links)

        return {
            'program': program['name'],
            'commission_rate': f"{program['commission_rate']}%",
            'total_links': len(program_links),
            'total_clicks': total_clicks,
            'total_conversions': total_conversions,
            'total_revenue': f"${total_revenue:.2f}",
            'avg_conversion_rate': f"{(total_conversions/total_clicks*100):.2f}%" if total_clicks > 0 else "0%"
        }

    def export_links(self, format: str = 'csv') -> str:
        """
        Export all affiliate links to file
        """
        filename = f"affiliate_links_{self.timestamp}.{format}"

        if format == 'csv':
            with open(filename, 'w', newline='') as f:
                writer = csv.DictWriter(
                    f,
                    fieldnames=['id', 'program_name', 'product_name', 'affiliate_link',
                               'clicks', 'conversions', 'revenue', 'created_date']
                )
                writer.writeheader()
                writer.writerows(self.links)

        elif format == 'json':
            with open(filename, 'w') as f:
                json.dump(self.links, f, indent=2)

        print(f"✅ Affiliate links exported to {filename}")
        return filename

    def generate_report(self):
        """
        Generate performance report
        """
        print("\n" + "="*80)
        print("📊 AFFILIATE PERFORMANCE REPORT")
        print("="*80)

        print(f"\nPrograms: {len(self.programs)}")
        print(f"Active Links: {len(self.links)}")
        print(f"Total Conversions: {sum(l['conversions'] for l in self.links)}")
        print(f"Total Revenue: ${sum(l['revenue'] for l in self.links):.2f}")

        print("\n" + "-"*80)
        print("Top Performing Links:")
        print("-"*80)

        sorted_links = sorted(self.links, key=lambda x: x['revenue'], reverse=True)[:5]

        for link in sorted_links:
            perf = self.get_link_performance(link['id'])
            print(f"\n  {link['product_name']}")
            print(f"    Revenue: {perf['revenue']}")
            print(f"    Conversions: {perf['conversions']}")
            print(f"    Conversion Rate: {perf['conversion_rate']}")

        print("\n" + "="*80)

"""

import json
from datetime import datetime, timedelta
from typing import Dict, List

class YouTubeAnalyticsDashboard:
    """Track video analytics and generate performance reports"""

    def __init__(self):
        self.videos = []

    def add_video(self, video_id: str, title: str, upload_date: str) -> Dict:
        """Add video to tracking"""

        video = {
            'video_id': video_id,
            'title': title,
            'upload_date': upload_date,
            'views': 0,
            'watch_time_hours': 0,
            'average_view_duration': 0,
            'click_through_rate': 0,
            'engagement_rate': 0,
            'likes': 0,
            'comments': 0,
            'shares': 0,
            'subscribers_gained': 0,
            'revenue': 0.0
        }

        self.videos.append(video)
        return video

    def update_video_metrics(self, video_id: str, **metrics):
        """Update video metrics"""
        video = next((v for v in self.videos if v['video_id'] == video_id), None)

        if video:
            video.update(metrics)
            print(f"✅ Updated metrics for: {video['title']}")

    def calculate_performance_score(self, video: Dict) -> float:
        """
        Calculate overall performance score (0-100)
        """
        score = 0

        # View performance (25 points)
        if video['views'] > 10000:
            score += 25
        elif video['views'] > 5000:
            score += 20
        elif video['views'] > 1000:
            score += 15
        elif video['views'] > 500:
            score += 10
        else:
            score += max(0, (video['views'] / 500) * 10)

        # Engagement metrics (25 points)
        engagement_rate = video.get('engagement_rate', 0)
        if engagement_rate > 5:
            score += 25
        elif engagement_rate > 3:
            score += 20
        elif engagement_rate > 1:
            score += 15
        elif engagement_rate > 0.5:
            score += 10
        else:
            score += max(0, (engagement_rate / 0.5) * 10)

        # Watch time efficiency (25 points)
        avd = video.get('average_view_duration', 0)
        if avd > 50:  # More than 50% watched
            score += 25
        elif avd > 40:
            score += 20
        elif avd > 25:
            score += 15
        elif avd > 10:
            score += 10
        else:
            score += max(0, (avd / 10) * 10)

        # Subscriber growth (25 points)
        subs_gained = video.get('subscribers_gained', 0)
        if subs_gained > 100:
            score += 25
        elif subs_gained > 50:
            score += 20
        elif subs_gained > 20:
            score += 15
        elif subs_gained > 5:
            score += 10
        else:
            score += max(0, (subs_gained / 5) * 10)

        return min(score, 100)

    def get_top_videos(self, limit: int = 5, metric: str = 'views') -> List[Dict]:
        """Get top performing videos"""
        sorted_videos = sorted(
            self.videos,
            key=lambda v: v.get(metric, 0),
            reverse=True
        )[:limit]

        return sorted_videos

    def calculate_channel_metrics(self) -> Dict:
        """Calculate channel-wide metrics"""

        if not self.videos:
            return {}

        total_views = sum(v['views'] for v in self.videos)
        total_hours = sum(v['watch_time_hours'] for v in self.videos)
        total_subs = sum(v['subscribers_gained'] for v in self.videos)
        avg_engagement = sum(v.get('engagement_rate', 0) for v in self.videos) / len(self.videos)
        total_revenue = sum(v['revenue'] for v in self.videos)

        return {
            'total_videos': len(self.videos),
            'total_views': total_views,
            'total_watch_hours': f"{total_hours:,.0f}",
            'subscribers_gained': total_subs,
            'average_engagement_rate': f"{avg_engagement:.2f}%",
            'total_revenue': f"${total_revenue:.2f}",
            'views_per_video': f"{(total_views / len(self.videos)):,.0f}",
            'avg_watch_time': f"{(total_hours / len(self.videos)):,.0f} hours"
        }

    def get_video_recommendations(self) -> List[Dict]:
        """Get optimization recommendations based on performance"""

        recommendations = []

        for video in self.videos:
            video_recs = []

            # Low CTR check
            if video['click_through_rate'] < 3:
                video_recs.append({
                    'issue': 'Low CTR',
                    'metric': f"{video['click_through_rate']:.2f}%",
                    'recommendation': 'Test new thumbnails - your current design isn\'t compelling',
                    'priority': 'High'
                })

            # Low engagement
            if video.get('engagement_rate', 0) < 1:
                video_recs.append({
                    'issue': 'Low Engagement',
                    'metric': f"{video['engagement_rate']:.2f}%",
                    'recommendation': 'Improve audio quality, pacing, or call-to-action',
                    'priority': 'High'
                })

            # High drop-off
            if video['average_view_duration'] < 25:
                video_recs.append({
                    'issue': 'High Drop-off',
                    'metric': f"{video['average_view_duration']:.0f}%",
                    'recommendation': 'Hook viewers faster in first 10 seconds',
                    'priority': 'High'
                })

            # Low subscriber conversion
            if video['views'] > 1000 and video['subscribers_gained'] < 5:
                video_recs.append({
                    'issue': 'Low Sub Conversion',
                    'metric': f"{(video['subscribers_gained']/video['views']*100):.2f}%",
                    'recommendation': 'Add verbal CTA to subscribe, improve channel branding',
                    'priority': 'Medium'
                })

            recommendations.append({
                'video_id': video['video_id'],
                'title': video['title'],
                'recommendations': video_recs
            })

        return recommendations

    def print_dashboard(self):
        """Print dashboard to console"""
        metrics = self.calculate_channel_metrics()

        print("\n" + "="*100)
        print("📊 YOUTUBE CHANNEL ANALYTICS DASHBOARD")
        print("="*100)

        print("\n📈 Channel Metrics:")
        for key, value in metrics.items():
            print(f"  {key}: {value}")

        print("\n" + "-"*100)
        print("🏆 Top 5 Videos:")
        print("-"*100)

        top_videos = self.get_top_videos(limit=5, metric='views')

        for i, video in enumerate(top_videos, 1):
            score = self.calculate_performance_score(video)
            print(f"\n  {i}. {video['title']}")
            print(f"     Views: {video['views']:,} | Watch Hours: {video['watch_time_hours']:,}")
            print(f"     CTR: {video['click_through_rate']:.2f}% | Engagement: {video.get('engagement_rate', 0):.2f}%")
            print(f"     Subs Gained: {video['subscribers_gained']} | Revenue: ${video['revenue']:.2f}")
            print(f"     Performance Score: {score:.0f}/100")
